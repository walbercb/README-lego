// 1. Mensagem de saudação inicial
console.log("Hello! I'm your coding fun fact guide!");

// 2. Criação das variáveis botName, botLocation e favoriteLanguage
const botName = "ByteBot";
const botLocation = "the Cloud";
const favoriteLanguage = "JavaScript";

// 3. Exibição do nome e localização do bot
console.log("My name is " + botName + " and I live on " + botLocation + ".");

// 4. Exibição da linguagem de programação favorita
console.log(`My favorite programming language is ${favoriteLanguage}.`);

// 5. Criação da variável codingFact (usando let) com o primeiro fato

// 6. Exibição do primeiro fato
console.log(codingFact);

// 7. Reatribuição da variável codingFact com o segundo fato
codingFact = `${favoriteLanguage} was originally named Mocha, then LiveScript.`;

// 8. Exibição do segundo fato
console.log(codingFact);

// 9. Reatribuição da variável codingFact com o terceiro fato
codingFact = `${favoriteLanguage} is used by both frontend and backend developers!`;

// 10. Exibição do terceiro fato
console.log(codingFact);

// 11. Declaração de despedida do bot
console.log("It was fun sharing these facts with you. Goodbye! - " + botName + " from " + botLocation + ".");
